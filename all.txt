% q1_dijkstra.m - Validacao da 1a Questao (Dijkstra, origem 0)
% Algoritmo: dijkstra() do material "algoritmos em MATLAB para Grafos"
%            + shortestpathtree/distances nativos do MATLAB (Lab. 01).
% Requer MATLAB R2016b ou superior (funcoes locais no final do script).
clear; close all; clc;
if ~exist('../figs','dir'), mkdir('../figs'); end
if ~exist('../out','dir'),  mkdir('../out');  end
arqOut = '../out/q1_dijkstra.txt';
if exist(arqOut,'file'), delete(arqOut); end
diary(arqOut);

%% Grafo (vertice k da prova = indice k+1 no MATLAB)
nomes = {'0' '1' '2' '3' '4' '5'};
W = zeros(6);
W(1,3) = 20;  W(1,4) = 40;  W(1,5) = 30;   % 0->2, 0->3, 0->4
W(2,3) = 10;                               % 1->2
W(3,5) = 15;                               % 2->4
W(4,5) = 10;  W(4,6) = 80;                 % 3->4, 3->5
W(5,2) = 82;  W(5,6) = 20;                 % 4->1, 4->5
% Aresta 5->1 sem peso na legenda: nao incluida.

%% Dijkstra (implementacao do material, com impressao das iteracoes)
[dist, pred] = dijkstra_tab(W, 1, nomes);

fprintf('\n--- Resultado final (origem 0) ---\n');
fprintf('%-8s %-8s %-12s %s\n','Destino','Custo','Antecessor','Caminho');
for v = 1:6
    p = caminho(pred, 1, v);
    if pred(v) == 0, a = '--'; else, a = nomes{pred(v)}; end
    fprintf('%-8s %-8g %-12s %s\n', nomes{v}, dist(v), a, strjoin(nomes(p),' -> '));
end

%% Validacao com funcoes nativas do MATLAB
G = digraph(W, nomes);
dNat = distances(G, '0');
fprintf('\nDistancias (distances nativo): %s\n', mat2str(dNat));
esperado = [0 112 20 40 30 50];                    % resolucao manual
fprintf('Confere com a resolucao manual: %s\n', simnao(isequal(dist, esperado) && isequal(dNat, esperado)));

%% Figura
figure('Name','Q1 - Dijkstra','Color','w');
Pl = plot(G,'EdgeLabel',G.Edges.Weight,'NodeFontSize',14,'NodeColor','k', ...
    'EdgeColor','b','LineWidth',1.5,'Marker','o','MarkerSize',8,'ArrowSize',12,'Layout','layered');

%% ===================== FUNCOES =====================
function [dist, pred] = dijkstra_tab(W, s, nomes)
    % dijkstra() do material + impressao da tabela de evolucao
    n = size(W, 1);
    dist = inf(1, n);  pred = zeros(1, n);  visitado = false(1, n);
    dist(s) = 0;
    fprintf('Iter | u | '); fprintf('%-9s', nomes{:}); fprintf('\n');
    for k = 1:n
        d = dist;  d(visitado) = inf;
        [m, u] = min(d);
        if isinf(m), break; end
        visitado(u) = true;
        for v = find(W(u, :))
            if ~visitado(v) && dist(u) + W(u, v) < dist(v)
                dist(v) = dist(u) + W(u, v);
                pred(v) = u;
            end
        end
        fprintf('%4d | %s | ', k, nomes{u});
        for v = 1:n
            if isinf(dist(v)), txt = 'inf';
            elseif pred(v) == 0, txt = sprintf('%g(-)', dist(v));
            else, txt = sprintf('%g(%s)', dist(v), nomes{pred(v)}); end
            fprintf('%-9s', txt);
        end
        fprintf('\n');
    end
end

function p = caminho(pred, s, v)
    p = v;
    while v ~= s && pred(v) ~= 0
        v = pred(v);  p = [v p];
    end
    if p(1) ~= s, p = []; end
end

function s = simnao(x)
    if x, s = 'SIM'; else, s = 'NAO'; end
end

function salvar_fig(f, arq)
    if exist('exportgraphics','file'), exportgraphics(f, arq, 'Resolution', 200);
    else, saveas(f, arq); end
end

%% ===================== FUNCOES =====================
function bellman_ford_tab(W, s, nomes)
    % Relaxamento por passadas: passada k = caminhos com ate k arestas
    n = size(W, 1);
    [I, J] = find(W);
    peso = W(sub2ind(size(W), I, J));
    dist = inf(1, n);  pred = zeros(1, n);  dist(s) = 0;
    fprintf('Passada | '); fprintf('%-8s', nomes{:}); fprintf('\n');
    imprime(0, dist, pred, nomes);
    for k = 1:n-1
        dAnt = dist;
        for e = 1:numel(I)
            if dAnt(I(e)) + peso(e) < dist(J(e))
                dist(J(e)) = dAnt(I(e)) + peso(e);
                pred(J(e)) = I(e);
            end
        end
        imprime(k, dist, pred, nomes);
    end
    mudou = false;
    for e = 1:numel(I)
        if dist(I(e)) + peso(e) < dist(J(e)), mudou = true; end
    end
    fprintf('Passada %d (verificacao): houve reducao? %s\n', n, simnao(mudou));
end

function imprime(k, dist, pred, nomes)
    fprintf('%7d | ', k);
    for v = 1:numel(dist)
        if isinf(dist(v)), txt = 'inf';
        elseif pred(v) == 0, txt = sprintf('%g(-)', dist(v));
        else, txt = sprintf('%g(%s)', dist(v), nomes{pred(v)}); end
        fprintf('%-8s', txt);
    end
    fprintf('\n');
end

function [dist, pred, cicloNeg] = bellman_ford(W, s)
    % Menores caminhos com pesos possivelmente negativos. O(V*E).
    n = size(W, 1);
    [I, J] = find(W);
    peso = W(sub2ind(size(W), I, J));
    dist = inf(1, n);  pred = zeros(1, n);  dist(s) = 0;
    for k = 1:n-1
        mudou = false;
        for e = 1:numel(I)
            if dist(I(e)) + peso(e) < dist(J(e))
                dist(J(e)) = dist(I(e)) + peso(e);
                pred(J(e)) = I(e);
                mudou = true;
            end
        end
        if ~mudou, break; end
    end
    cicloNeg = false;
    for e = 1:numel(I)
        if dist(I(e)) + peso(e) < dist(J(e))
            cicloNeg = true;  break;
        end
    end
end

function [fluxoMax, F] = ford_fulkerson(C, s, t)
    % Fluxo maximo: caminhos aumentantes por busca em profundidade no residual
    n = size(C, 1);  R = C;  fluxoMax = 0;
    while true
        pai = zeros(1, n);  pai(s) = s;  pilha = s;
        while ~isempty(pilha) && pai(t) == 0
            u = pilha(end);  pilha(end) = [];
            for v = find(R(u, :) > 0)
                if pai(v) == 0
                    pai(v) = u;  pilha(end+1) = v;
                end
            end
        end
        if pai(t) == 0, break; end
        gargalo = inf;  v = t;
        while v ~= s
            u = pai(v);  gargalo = min(gargalo, R(u, v));  v = u;
        end
        v = t;
        while v ~= s
            u = pai(v);
            R(u, v) = R(u, v) - gargalo;
            R(v, u) = R(v, u) + gargalo;
            v = u;
        end
        fluxoMax = fluxoMax + gargalo;
    end
    F = max(C - R, 0);
end

function vis = bfs_residual(R, s)

    n = size(R, 1);  vis = false(1, n);
    fila = zeros(1, n);  cab = 1;  cau = 1;
    fila(1) = s;  vis(s) = true;
    while cab <= cau
        v = fila(cab);  cab = cab + 1;
        for w = find(R(v, :) > 0)
            if ~vis(w)
                vis(w) = true;  cau = cau + 1;  fila(cau) = w;
            end
        end
    end
end

function [dist, pred] = dijkstra_tab(W, s, nomes)
n = size(W, 1);
dist = inf(1, n);  pred = zeros(1, n);  visitado = false(1, n);
dist(s) = 0;
fprintf('Iter | u | '); fprintf('%-9s', nomes{:}); fprintf('\n');

    d = dist;  d(visitado) = inf;
    [m, u] = min(d);
    if isinf(m), break; end
    visitado(u) = true;
    for v = find(W(u, :))
        if ~visitado(v) && dist(u) + W(u, v) < dist(v)
            dist(v) = dist(u) + W(u, v);
            pred(v) = u;
        end
    end
    fprintf('%4d | %s | ', k, nomes{u});

function bellman_ford_tab(W, s, nomes)

% ===================== SUBFUNCOES =====================
function fluxoMax = ford_fulkerson(C, s, t)
n = size(C, 1);  R = C;  fluxoMax = 0;
while true
    pai = zeros(1, n);  pai(s) = s;  pilha = s;
    while ~isempty(pilha) && pai(t) == 0
        u = pilha(end);  pilha(end) = [];
        for v = find(R(u, :) > 0)
            if pai(v) == 0
                pai(v) = u;  pilha(end+1) = v; %#ok<AGROW>
            end
        end
    end
    if pai(t) == 0, break; end
    gargalo = inf;  v = t;
    while v ~= s
        u = pai(v);  gargalo = min(gargalo, R(u, v));  v = u;
    end
    v = t;
    while v ~= s
        u = pai(v);
        R(u, v) = R(u, v) - gargalo;
        R(v, u) = R(v, u) + gargalo;
        v = u;
    end
    fluxoMax = fluxoMax + gargalo;
end
end

function vis = bfs_residual(R, s)
n = size(R, 1);  vis = false(1, n);
fila = zeros(1, n);  cab = 1;  cau = 1;
fila(1) = s;  vis(s) = true;
while cab <= cau
    v = fila(cab);  cab = cab + 1;
    for w = find(R(v, :) > 0)
        if ~vis(w)
            vis(w) = true;  cau = cau + 1;  fila(cau) = w;
        end
    end
end
end
